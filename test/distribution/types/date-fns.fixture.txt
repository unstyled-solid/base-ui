import { TemporalAdapterDateFns } from 'baseui-solid2/internals/temporal-adapter-date-fns';

import { enUS } from 'date-fns/locale/en-US';

const date: Date = adapter.date('2026-10-05', 'UTC');
const nil: null = adapter.date(null, 'UTC');
const registered: TemporalSupportedObjectLookup['date-fns'] = date;
void [nil, registered];
